# Load required libraries
library(rxode2)
library(tidyverse)
library(data.table)

# ============================================================================
# 1. PHARMACOKINETIC PARAMETERS
# ============================================================================

# Population PK Parameters for Palbociclib
# Source: Published clinical trial data (Cassier et al., Infante et al.)
# Units: CL in L/h, V in L

pk_params <- list(
  # Central tendency parameters
  CL = 63,              # Clearance (L/h) - population mean
  V = 2710,             # Volume of distribution (L) - population mean
  
  # Inter-individual variability (IIV) - coefficient of variation
  CL_iiv = 0.35,        # 35% IIV on clearance
  V_iiv = 0.40,         # 40% IIV on volume
  
  # Absorption parameters
  ka = 0.5,             # Absorption rate constant (1/h)
  F = 0.68,             # Bioavailability
  
  # Residual variability
  prop_error = 0.25,    # Proportional error (25%)
  add_error = 10        # Additive error (ng/mL)
)

# Dose specifications
dose_params <- list(
  dose_standard = 125,  # Standard palbociclib dose (mg)
  dose_reduced = 100,   # Reduced dose for TDM (mg)
  dose_interval = 24,   # Dosing interval (hours) - daily dosing
  
  # Dosing schedule: 21 days on, 7 days off
  cycle_length = 28,    # 28-day cycle
  dosing_days = 21,     # Days 1-21: drug given
  rest_days = 7         # Days 22-28: no dosing
)

# Target concentration for TDM decision
tdm_params <- list(
  tdm_sampling_day = 15,     # Sample on Day 15 of cycle
  tdm_threshold = 100,       # Threshold concentration (ng/mL)
  tdm_threshold_unit = "ng/mL"
  # Rule: if Cmin > 100 ng/mL on Day 15, reduce dose to 100 mg starting Cycle 2
)

# ============================================================================
# 2. PHARMACODYNAMIC PARAMETERS - NEUTROPENIA EXPOSURE-RESPONSE
# ============================================================================

# Grade 3/4 Neutropenia Risk Model
# Exposure-response relationship: Cmin vs probability of neutropenia
# Source: Clinical trial data and published exposure-response analyses

pd_params <- list(
  # Baseline neutropenia risk at standard dosing
  baseline_neutropenia_risk = 0.66,  # 66% at standard dose (published data)
  
  # Exposure-response parameters (Sigmoid Emax model)
  E0 = 0.10,              # Baseline toxicity risk (10% background)
  Emax = 0.90,            # Maximum effect (90%)
  EC50 = 85,              # Concentration producing 50% effect (ng/mL)
  gamma = 1.5,            # Hill coefficient (sigmoidicity)
  
  # Risk relationship: higher exposure = higher neutropenia risk
  # Mathematical form: Risk = E0 + (Emax-E0) * (Cmin^gamma) / (EC50^gamma + Cmin^gamma)
  
  # Alternative simple linear model for sensitivity
  slope_linear = 0.004,   # Risk increases ~0.4% per ng/mL increase
  intercept_linear = 0.05
)

# ============================================================================
# 3. COST PARAMETERS
# ============================================================================

cost_params <- list(
  # Hospitalization for neutropenia
  hospitalization_cost = 22839,    # USD per event (literature value)
  
  # TDM program costs
  tdm_assay_cost = 350,            # Cost per TDM assessment (USD)
  tdm_frequency = 1,               # TDM assessment per cycle per patient
  
  # Drug costs (for reference, not included in base analysis)
  palbociclib_cost_per_dose = 50,  # Approximate cost per 125mg dose
  
  # Avoided costs from dose reduction
  dose_reduction_saves = 50         # Savings per dose (25 mg difference)
)

# ============================================================================
# 4. PATIENT POPULATION PARAMETERS
# ============================================================================

population_params <- list(
  n_patients = 1000,         # Simulation cohort size
  
  # Demographic distributions
  age_mean = 62,             # Mean age (years)
  age_sd = 8,                # SD of age
  
  # Weight-based dose adjustments
  weight_mean = 72,          # Mean weight (kg)
  weight_sd = 12,            # SD of weight
  
  # Renal function
  creatinine_mean = 0.95,    # Mean serum creatinine (mg/dL)
  creatinine_sd = 0.25,
  
  # Hepatic function
  bilirubin_mean = 0.8,      # Mean total bilirubin (mg/dL)
  bilirubin_sd = 0.3
)

# ============================================================================
# 5. SIMULATION PARAMETERS
# ============================================================================

simulation_params <- list(
  # Number of treatment cycles
  n_cycles = 4,              # Simulate 4 cycles (112 days)
  
  # Sample time points
  sample_times = c(0, 1, 2, 4, 8, 12, 24, 36, 48),  # Hours after dose
  
  # Seed for reproducibility
  set_seed = 12345,
  
  # Tolerance for ODE solver
  atol = 1e-6,
  rtol = 1e-4
)

# ============================================================================
# 6. ANALYSIS PARAMETERS
# ============================================================================

analysis_params <- list(
  # Number of Monte Carlo samples (for sensitivity analysis)
  n_monte_carlo = 10000,
  
  # Parameter uncertainty ranges (±%)
  param_uncertainty = 0.30,  # 30% coefficient of variation
  
  # Sensitivity analysis parameters
  sensitivity_ranges = list(
    CL_range = c(40, 80),         # Clearance variation (L/h)
    V_range = c(1800, 3600),      # Volume variation (L)
    risk_range = c(0.50, 0.80)    # Baseline risk variation
  )
)

# ============================================================================
# 7. VALIDATION PARAMETERS - TARGET RESULTS
# ============================================================================

# Expected simulation outputs (for validation)
expected_results <- list(
  baseline_neutropenia_risk = 0.224,      # 22.4% at standard dose
  tdm_neutropenia_risk = 0.183,           # 18.3% with TDM
  baseline_total_cost = 5104000,          # $5,104,000 for 1,000 patients
  tdm_total_cost = 4145000,               # $4,145,000 with TDM
  annual_savings = 586000,                # $586,000 savings
  
  # Sensitivity analysis range
  savings_min = 400000,                   # Minimum savings (sensitivity)
  savings_max = 750000                    # Maximum savings (sensitivity)
)

# ============================================================================
# 8. SAVE ALL PARAMETERS AS R OBJECT
# ============================================================================

# Create comprehensive parameter list
all_params <- list(
  pk = pk_params,
  dose = dose_params,
  tdm = tdm_params,
  pd = pd_params,
  cost = cost_params,
  population = population_params,
  simulation = simulation_params,
  analysis = analysis_params,
  expected = expected_results
)

# Save to RDS for use in other scripts
saveRDS(all_params, "data/parameters.rds")

# Print summary to console
cat("\n")
cat("==========================================================================\n")
cat("PALBOCICLIB TDM SIMULATION - PARAMETERS LOADED SUCCESSFULLY\n")
cat("==========================================================================\n")
cat("\nParameter Summary:\n")
cat("- PK Model: 2-compartment (oral absorption, linear elimination)\n")
cat("- Population size: ", population_params$n_patients, " patients\n")
cat("- Treatment duration: ", simulation_params$n_cycles, " cycles (", 
    simulation_params$n_cycles * 28, " days)\n")
cat("- TDM decision point: Day", tdm_params$tdm_sampling_day, 
    " (threshold = ", tdm_params$tdm_threshold, " ng/mL)\n")
cat("- Hospitalization cost per event: $", 
    format(cost_params$hospitalization_cost, big.mark=","), "\n")
cat("\nExpected Results:\n")
cat("- Baseline neutropenia risk: ", 
    round(expected_results$baseline_neutropenia_risk*100, 1), "%\n")
cat("- TDM-guided neutropenia risk: ", 
    round(expected_results$tdm_neutropenia_risk*100, 1), "%\n")
cat("- Risk reduction: ", 
    round((expected_results$baseline_neutropenia_risk - 
           expected_results$tdm_neutropenia_risk)*100, 1), "%\n")
cat("- Annual savings (1,000 patients): $", 
    format(expected_results$annual_savings, big.mark=","), "\n")
cat("==========================================================================\n\n")
